import RPi.GPIO as GPIO
import time

gas = 11
relay = [40, 38, 36, 32]

def setup():
    GPIO.setmode(GPIO.BOARD)
    GPIO.setwarnings(False)
    GPIO.setup(gas, GPIO.IN)
    for r in relay:
        GPIO.setup(r, GPIO.OUT)

def read_gas():
    while True:
        gas_state = GPIO.input(gas)

        if gas_state == 0:
            print("Gas detected")
            for r in relay:
                GPIO.output(r, True)
        else:
            print("No Gas")
            for r in relay:
                GPIO.output(r, False)

        time.sleep(0.3)

def destroy():
    GPIO.cleanup()

if __name__ == "__main__":
    try:
        setup()
        read_gas()
    except KeyboardInterrupt:
        destroy()
