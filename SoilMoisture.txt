import RPi.GPIO as GPIO
import time

channel = 11

GPIO.setmode(GPIO.BOARD)
GPIO.setup(channel, GPIO.IN)

try:
    while True:
        if GPIO.input(channel) == 0:
            print("Moisture Detected")
        else:
            print("No Moisture")

        time.sleep(2)

except KeyboardInterrupt:
    GPIO.cleanup()
