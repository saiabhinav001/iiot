import RPi.GPIO as GPIO
import time

sensor = 17

GPIO.setmode(GPIO.BCM)
GPIO.setup(sensor, GPIO.IN)

try:
    while True:
        if GPIO.input(sensor) == 0:
            print("Rain Detected")
        else:
            print("No Rain")

        time.sleep(0.5)

except KeyboardInterrupt:
    GPIO.cleanup()
